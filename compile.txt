финальная агрессивная компиляция с logo
g++ -O3 -municode -mwindows main.cpp resources.o -o CustomDesktopOverlay.exe -static -static-libgcc -static-libstdc++ -lshell32 -lcomctl32 -lgdiplus -lole32 -lwindowscodecs -s -Wl,--strip-all -ffunction-sections -fdata-sections -Wl,--gc-sections -fno-exceptions -fno-rtti -fno-asynchronous-unwind-tables -fno-ident -flto -fomit-frame-pointer -fvisibility=hidden -pipe -Wl,--nxcompat -Wl,--dynamicbase
